<h1>Facebook</h1>
<ul>
<li> "username" site:facebook.com inurl:posts</li>
</ul>

<h1>Instagram</h1>
<ul>
  <li> site:instagram.com "username"</li>
  <li> site:instagram.com "@username"</li>
  <li> "username" "instagram.com/p/" site:instagram.com </li>
  <li> "username" site:instagram.com inurl:"/p/" </li>
  <li> "username" site:instagram.com inurl:/p/ </li>
</ul>
